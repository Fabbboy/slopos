# check_stack_sizes allowlist — variant: dev
#
# Frames above STACK_SIZE_THRESHOLD (2048 B) that are permitted, each capped
# at its measured size so growth still fails.
#
#     scripts/check_stack_sizes.sh --variant dev --emit-allowlist builddir/kernel-dev.elf
#
# Every entry must match at least one over-threshold frame: a dead exemption
# fails the gate, so a frame that shrinks hands its exemption back. No entry
# may exceed the 4096 B guard page — such a frame is a bug to fix.

# min-records — the ELF must carry at least this many .stack_sizes entries;
# llvm-readobj prints an empty `StackSizes [ ]` and exits 0 without the
# section. Measured 2026-07-31: dev 31234, tests 44732, release 3830. The dev
# floor sits deliberately above release's count, so `--variant dev` against a
# release ELF trips here. Keep that relation when adjusting it.
min-records 20000

# few | many | any. A kernel/tests image links ~2742 104-byte .test_registry
# entries; dev and release link 6. Cannot separate dev from release — that is
# min-records' job.
expect-test-registry few
